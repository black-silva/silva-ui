# Popover

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { Button, Popover, PopoverClose, PopoverContent, PopoverTrigger, Text } from "silva-ui"
import './styles.css'

function previewControl(event: Event) {
  return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]')
}

function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event)
  if (!root) return
  root.dataset.open = String(open)
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => {
    layer.hidden = !open
    layer.setAttribute('data-state', open ? 'open' : 'closed')
  })
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]')
  if (toast) toast.hidden = !open
}

<div class="preview-menu preview-interactive" data-preview-control>
  <Popover>
    <PopoverTrigger><Button variant="outline" onClick={(event) => setPreviewOpen(event, true)}>Open popover</Button></PopoverTrigger>
    <PopoverContent class="preview-popout preview-popover" data-preview-layer data-state="closed" style={{ position: 'static' }}>
      <Text>A little context, right when you need it.</Text>
      <PopoverClose onClick={(event) => setPreviewOpen(event, false)}>Close</PopoverClose>
    </PopoverContent>
  </Popover>
</div>
```
