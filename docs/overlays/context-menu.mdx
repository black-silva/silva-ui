# ContextMenu

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { Button, ContextMenu, ContextMenuContent, ContextMenuGroup, ContextMenuItem, ContextMenuLabel, ContextMenuSeparator, ContextMenuTrigger, YStack } from "silva-ui"

function previewControl(event: Event) { return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]'); }
function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event);
  if (!root) return;
  root.dataset.open = String(open);
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => { layer.hidden = !open; layer.setAttribute('data-state', open ? 'open' : 'closed'); });
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]');
  if (toast) toast.hidden = !open;
}

const Interactive = ({ children, menu = false }: { children: any; menu?: boolean }) => (
  <YStack data-preview-control width="64" gap="small" alignItems="center" justifyContent="center" minHeight={menu ? '40' : undefined} position={menu ? 'relative' : undefined}>{children}</YStack>
);

const variant = 0;

<Interactive menu><ContextMenu><ContextMenuTrigger onContextMenu={(event) => { event.preventDefault(); setPreviewOpen(event, true); }} onClick={(event) => setPreviewOpen(event, true)}><Button variant="outline">{variant ? 'Right-click for actions' : 'Right-click this card'}</Button></ContextMenuTrigger><ContextMenuContent data-preview-layer data-state="closed">{variant ? <><ContextMenuLabel>Actions</ContextMenuLabel><ContextMenuGroup><ContextMenuItem onClick={(event) => setPreviewOpen(event, false)}>Edit</ContextMenuItem></ContextMenuGroup><ContextMenuSeparator /><ContextMenuItem onClick={(event) => setPreviewOpen(event, false)}>Delete</ContextMenuItem></> : <ContextMenuItem onClick={(event) => setPreviewOpen(event, false)}>Edit</ContextMenuItem>}</ContextMenuContent></ContextMenu></Interactive>
```
