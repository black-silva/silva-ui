# Accordion

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { Accordion, AccordionContent, AccordionItem, AccordionTrigger } from "silva-ui"

function toggleAccordion(event: Event) {
  const trigger = event.currentTarget as HTMLElement;
  const item = trigger.closest<HTMLElement>('[data-preview-accordion-item]');
  if (!item) return;
  const open = trigger.getAttribute('data-state') !== 'open';
  trigger.setAttribute('data-state', open ? 'open' : 'closed');
  trigger.setAttribute('aria-expanded', String(open));
  const content = item.querySelector<HTMLElement>('[data-preview-accordion-content]');
  if (content) { content.hidden = !open; content.setAttribute('data-state', open ? 'open' : 'closed'); }
}

const variant = 0;

<Accordion width="64"><AccordionItem value={variant ? 'two' : 'one'} data-preview-accordion-item><AccordionTrigger data-state="closed" onClick={toggleAccordion}>{variant ? 'Is it typed?' : 'What is Silva UI?'}</AccordionTrigger><AccordionContent data-preview-accordion-content data-state="closed" hidden>{variant ? 'Yes, it is TypeScript-first.' : 'Composable UI for Hono JSX.'}</AccordionContent></AccordionItem></Accordion>
```
