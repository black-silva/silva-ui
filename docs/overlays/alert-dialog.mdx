# AlertDialog

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogOverlay, AlertDialogTitle, Button } from "silva-ui"
import './styles.css'

function previewControl(event: Event) {
  return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]')
}

function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event)
  if (!root) return
  root.dataset.open = String(open)
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => {
    layer.hidden = !open
    layer.setAttribute('data-state', open ? 'open' : 'closed')
  })
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]')
  if (toast) toast.hidden = !open
}

<div class="preview-interactive" data-preview-control>
  <Button onClick={(event) => setPreviewOpen(event, true)}>Delete project</Button>
  <AlertDialog open={false}>
    <AlertDialogOverlay class="preview-overlay" data-preview-layer hidden hidden onClick={(event) => setPreviewOpen(event, false)} />
    <AlertDialogContent class="preview-modal preview-alert" data-preview-layer hidden>
      <AlertDialogHeader><AlertDialogTitle>Delete project?</AlertDialogTitle><AlertDialogDescription>This cannot be undone.</AlertDialogDescription></AlertDialogHeader>
      <AlertDialogFooter><AlertDialogCancel onClick={(event) => setPreviewOpen(event, false)}>Cancel</AlertDialogCancel><AlertDialogAction onClick={(event) => setPreviewOpen(event, false)}>Delete</AlertDialogAction></AlertDialogFooter>
    </AlertDialogContent>
  </AlertDialog>
</div>
```
