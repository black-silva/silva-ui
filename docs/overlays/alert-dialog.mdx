# AlertDialog

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogOverlay, AlertDialogTitle, Button, YStack } from "silva-ui"

function previewControl(event: Event) { return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]'); }
function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event);
  if (!root) return;
  root.dataset.open = String(open);
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => { layer.hidden = !open; layer.setAttribute('data-state', open ? 'open' : 'closed'); });
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]');
  if (toast) toast.hidden = !open;
}

const Interactive = ({ children, menu = false }: { children: any; menu?: boolean }) => (
  <YStack data-preview-control width="64" gap="small" alignItems="center" justifyContent="center" minHeight={menu ? '40' : undefined} position={menu ? 'relative' : undefined}>{children}</YStack>
);

const variant = 0;

<Interactive><Button onClick={(event) => setPreviewOpen(event, true)}>{variant ? 'Discard draft' : 'Delete project'}</Button><AlertDialog open={false}><AlertDialogOverlay data-preview-layer hidden onClick={(event) => setPreviewOpen(event, false)} /><AlertDialogContent data-preview-layer hidden><AlertDialogHeader><AlertDialogTitle>{variant ? 'Discard changes?' : 'Delete project?'}</AlertDialogTitle><AlertDialogDescription>{variant ? 'Keep your current draft.' : 'This cannot be undone.'}</AlertDialogDescription></AlertDialogHeader><AlertDialogFooter><AlertDialogCancel onClick={(event) => setPreviewOpen(event, false)}>{variant ? 'Keep editing' : 'Cancel'}</AlertDialogCancel><AlertDialogAction onClick={(event) => setPreviewOpen(event, false)}>{variant ? 'Discard' : 'Delete'}</AlertDialogAction></AlertDialogFooter></AlertDialogContent></AlertDialog></Interactive>
```

  </Tab>
</Tabs>
