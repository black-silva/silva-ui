# Menu

```tsx
import { Button, Menu, MenuContent, MenuItem, MenuTrigger } from "silva-ui"
import './styles.css'

function previewControl(event: Event) {
  return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]')
}

function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event)
  if (!root) return
  root.dataset.open = String(open)
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => {
    layer.hidden = !open
    layer.setAttribute('data-state', open ? 'open' : 'closed')
  })
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]')
  if (toast) toast.hidden = !open
}

<div class="preview-menu preview-interactive" data-preview-control>
  <Menu>
    <MenuTrigger><Button variant="outline" onClick={(event) => setPreviewOpen(event, true)}>Open menu</Button></MenuTrigger>
    <MenuContent class="preview-popout" data-preview-layer data-state="closed" style={{ position: 'static' }}><MenuItem onClick={(event) => setPreviewOpen(event, false)}>Edit</MenuItem></MenuContent>
  </Menu>
</div>
```

function setPreviewOpen(event: Event, open: boolean) {
