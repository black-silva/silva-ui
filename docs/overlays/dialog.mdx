# Dialog

Two rendered compositions using exported Silva UI components. The code beside each preview is the implementation rendered by that preview. Components that need application state use an explicit static state here.

```tsx
import { Button, Dialog, DialogClose, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogOverlay, DialogTitle } from "silva-ui"
import './styles.css'

function previewControl(event: Event) {
  return (event.currentTarget as HTMLElement).closest<HTMLElement>('[data-preview-control]')
}

function setPreviewOpen(event: Event, open: boolean) {
  const root = previewControl(event)
  if (!root) return
  root.dataset.open = String(open)
  root.querySelectorAll<HTMLElement>('[data-preview-layer]').forEach((layer) => {
    layer.hidden = !open
    layer.setAttribute('data-state', open ? 'open' : 'closed')
  })
  const toast = root.querySelector<HTMLElement>('[data-preview-toast]')
  if (toast) toast.hidden = !open
}

<div class="preview-interactive" data-preview-control>
  <Button onClick={(event) => setPreviewOpen(event, true)}>Publish changes</Button>
  <Dialog open={false}>
    <DialogOverlay class="preview-overlay" data-preview-layer onClick={(event) => setPreviewOpen(event, false)} />
    <DialogContent class="preview-modal" data-preview-layer hidden>
      <DialogHeader><DialogTitle>Publish changes?</DialogTitle><DialogDescription>Your documentation will update.</DialogDescription></DialogHeader>
      <DialogFooter><DialogClose onClick={(event) => setPreviewOpen(event, false)}>Cancel</DialogClose><Button onClick={(event) => setPreviewOpen(event, false)}>Publish</Button></DialogFooter>
    </DialogContent>
  </Dialog>
</div>
```
